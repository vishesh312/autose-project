def _canonicalize_tool_call_args(args: Any) -> Any:
    if isinstance(args, dict):
        return {k: _canonicalize_tool_call_args(v) for k, v in sorted(args.items(), key=lambda kv: str(kv[0]))}
    if isinstance(args, (list, tuple)):
        return [_canonicalize_tool_call_args(v) for v in args]
    if isinstance(args, str):
